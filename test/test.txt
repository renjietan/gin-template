package main

import (
	"fmt"
	"os"
	"os/signal"
	"sync"
	"sync/atomic"
	"syscall"
	"time"
)

type Message struct {
	ID      int
	Content Content
	done    atomic.Bool
}

type Content struct {
	ID      int
	Content string
}

var data = make(chan *Message, 100)
var mu sync.Mutex
var data2 map[string]*sync.Map

func main() {
	data2 = make(map[string]*sync.Map)
	var quit_chan = make(chan os.Signal, 1)
	signal.Notify(quit_chan, os.Interrupt, syscall.SIGTERM)
	var wg sync.WaitGroup
	for v := range 3 {
		wg.Add(1)
		go func(index int) {

			key := fmt.Sprintf("%d", index)
			defer wg.Done()
			msg := &Message{
				ID: index,
				Content: Content{
					ID:      index,
					Content: key,
				},
			}
			msg.done.Store(true)
			data <- msg
			if index%2 == 0 {
				mu.Lock()
				time.Sleep(time.Duration((index+1)*3) * time.Second)
				if _, ok := data2[key]; !ok {
					data2[key] = &sync.Map{}
				}
				data2[key].Store(index, Message{
					ID: index,
					Content: Content{
						ID:      index,
						Content: fmt.Sprintf("data2-%d", index),
					},
				})
				mu.Unlock()
			}
		}(v)
	}

	wg.Add(1)
	go func() {
		defer wg.Done()
		for {
			select {
			case <-data:
				mu.Lock()
				fmt.Println("==========================================")
				for _, v := range data2 {
					v.Range(func(key, value interface{}) bool {
						fmt.Printf("key:%v, value:%#v\n", key, value)
						return true
					})
				}
				mu.Unlock()
			case <-quit_chan:
				fmt.Println("quit")
				return
			default:
				fmt.Println("================== default ========================")
				mu.Lock()
				for _, v := range data2 {
					v.Range(func(key, value interface{}) bool {
						fmt.Printf("key:%v, value:%#v\n", key, value)
						return true
					})
				}
				mu.Unlock()
				// fmt.Println("default")
				// close(data)
				// return
				// close(data)
			}
			time.Sleep(1 * time.Second)
		}
	}()
	wg.Wait()
}
